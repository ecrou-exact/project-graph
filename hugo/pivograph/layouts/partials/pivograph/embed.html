{{- /* A graph on a page: the text (works without JavaScript), a JSON file built by Hugo, and, when JavaScript runs, the interactive graph in the Pivograph app. Arguments: dict page the current page (required) section / data where the graph comes from (see document.html) app the Pivograph app (default: params.pivograph.app, else the public app) height of the interactive graph (default "75vh") sidebar show the app's side panel (default false) tags show the tag pills on the graph (default true) picture false to leave out the picture drawn by Hugo (shown without JavaScript) image a picture of your own (Export → Picture in the app), shown instead of Hugo's title accessible name of the graph (default: the graph's title) level heading level of the text's sections (default 3) text false to leave out the text (not recommended: it is the only version without JavaScript) */ -}} {{- $page := .page -}} {{- $doc := partial "pivograph/document.html" . -}} {{- $app := .app | default site.Params.pivograph.app | default "https://ecrou-exact.github.io/project-graph/" -}} {{- if not (strings.HasSuffix $app "/") -}}{{- $app = printf "%s/" $app -}}{{- end -}} {{- $key := .data | default (.section | default $page.CurrentSection.Path) | default "graph" | urlize | default "graph" -}} {{- $json := resources.FromString (printf "pivograph/%s.json" $key) (jsonify (dict "indent" " ") $doc) -}} {{- $title := .title | default $doc.meta.title | default "Graph" -}} {{- $sidebar := cond (eq .sidebar true) "1" "0" -}} {{- $tags := cond (eq .tags false) "0" "1" -}} {{- /* The script and the styles.
