(reflect.IsSlice .) -}}{{- $scalars = false -}}{{- end -}}{{- end -}} {{- $parts := slice -}} {{- range $v -}}{{- $parts = $parts | append (partial "pivograph/value.html" .) -}}{{- end -}} {{- if $scalars -}} {{- $out = delimit $parts ", " -}} {{- else -}} {{- $out = printf "
    %s
" (delimit (apply $parts "printf" "
  • %s
    • " ".") "") -}} {{- end -}} {{- else if eq (printf "%T" $v) "bool" -}} {{- $out = cond $v "yes" "no" -}} {{- else -}} {{- $s := string $v -}} {{- if findRE `^(https?://|mailto:)\S+$` $s -}} {{- $out = printf `%s` (htmlEscape $s) (htmlEscape (replaceRE `^(https?://|mailto:)` "" $s)) -}} {{- else -}} {{- $out = htmlEscape $s -}} {{- end -}} {{- end -}} {{- return $out -}}
